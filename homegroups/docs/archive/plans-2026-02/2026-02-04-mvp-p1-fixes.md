# Fix 3: Meeting Cancellation Notifications

**Problem:** When a meeting is cancelled, members are not notified, causing them to show up to cancelled meetings.

**Impact:** Retention - Members blame the app when they miss cancellation notices

## Task 3.1: Create Cloud Function for Meeting Change Notifications

**Files:**
- Create: `functions/src/triggers/firestore/onMeetingInstanceUpdate.ts`
- Modify: `functions/src/index.ts`

**Step 1: Create the trigger function**

```typescript
import * as functions from 'firebase-functions';
import * as admin from 'firebase-admin';

const db = admin.firestore();
const messaging = admin.messaging();

export const onMeetingInstanceUpdate = functions.firestore
  .document('meetingInstances/{instanceId}')
  .onUpdate(async (change, context) => {
    const before = change.before.data();
    const after = change.after.data();

    // Check if meeting was just cancelled
    const wasCancelled = !before.isCancelled && after.isCancelled;

    // Check if location changed significantly
    const locationChanged = before.location !== after.location ||
      before.address !== after.address;

    // Check if time changed
    const timeChanged = before.scheduledAt?.toMillis() !== after.scheduledAt?.toMillis();

    if (!wasCancelled && !locationChanged && !timeChanged) {
      return null; // No significant change
    }

    const groupId = after.groupId;
    const meetingName = after.name;

    // Get all group members with push notifications enabled
    const membersSnapshot = await db
      .collection('members')
      .where('groupId', '==', groupId)
      .get();

    const memberIds = membersSnapshot.docs.map(doc => doc.data().userId);

    // Get user FCM tokens
    const usersSnapshot = await db
      .collection('users')
      .where(admin.firestore.FieldPath.documentId(), 'in', memberIds.slice(0, 10)) // Firestore limit
      .get();

    const tokens: string[] = [];
    usersSnapshot.docs.forEach(doc => {
      const userData = doc.data();
      if (userData.notificationSettings?.meetings !== false &&
          userData.notificationSettings?.allowPushNotifications !== false &&
          userData.fcmTokens?.length > 0) {
        tokens.push(...userData.fcmTokens);
      }
    });

    if (tokens.length === 0) {
      console.log('No FCM tokens found for group members');
      return null;
    }

    // Build notification
    let title: string;
    let body: string;

    if (wasCancelled) {
      title = `Meeting Cancelled`;
      body = `${meetingName} has been cancelled.`;
      if (after.instanceNotice) {
        body += ` Note: ${after.instanceNotice}`;
      }
    } else if (timeChanged) {
      const newTime = after.scheduledAt.toDate().toLocaleTimeString('en-US', {
        hour: 'numeric',
        minute: '2-digit',
      });
      title = `Meeting Time Changed`;
      body = `${meetingName} has been rescheduled to ${newTime}.`;
    } else if (locationChanged) {
      title = `Meeting Location Changed`;
      body = `${meetingName} has moved to ${after.location || after.address}.`;
    }

    // Send notifications
    const message: admin.messaging.MulticastMessage = {
      tokens,
      notification: {
        title,
        body,
      },
      data: {
        type: 'meeting_change',
        groupId,
        instanceId: context.params.instanceId,
      },
      apns: {
        payload: {
          aps: {
            sound: 'default',
            badge: 1,
          },
        },
      },
      android: {
        priority: 'high',
        notification: {
          sound: 'default',
          priority: 'high',
        },
      },
    };

    try {
      const response = await messaging.sendEachForMulticast(message);
      console.log(`Sent ${response.successCount} meeting change notifications`);

      // Clean up invalid tokens
      response.responses.forEach((resp, idx) => {
        if (!resp.success && resp.error?.code === 'messaging/registration-token-not-registered') {
          // Token is invalid, could clean up here
          console.log(`Invalid token: ${tokens[idx]}`);
        }
      });
    } catch (error) {
      console.error('Error sending meeting change notifications:', error);
    }

    return null;
  });
```

// Replace the member/user fetching with batched approach:

async function getGroupMemberTokens(groupId: string): Promise<string[]> {
  const membersSnapshot = await db
    .collection('members')
    .where('groupId', '==', groupId)
    .get();

  const memberIds = membersSnapshot.docs.map(doc => doc.data().userId);
  const tokens: string[] = [];

  // Batch in groups of 10 (Firestore 'in' query limit)
  for (let i = 0; i < memberIds.length; i += 10) {
    const batch = memberIds.slice(i, i + 10);
    const usersSnapshot = await db
      .collection('users')
      .where(admin.firestore.FieldPath.documentId(), 'in', batch)
      .get();

    usersSnapshot.docs.forEach(doc => {
      const userData = doc.data();
      if (userData.notificationSettings?.meetings !== false &&
          userData.notificationSettings?.allowPushNotifications !== false &&
          userData.fcmTokens?.length > 0) {
        tokens.push(...userData.fcmTokens);
      }
    });
  }

  return tokens;
}
